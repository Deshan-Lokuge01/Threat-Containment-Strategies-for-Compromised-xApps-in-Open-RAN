a1mediator and e2mgr
---------------
first restart coredns

kubectl rollout restart deployment/coredns -n kube-system
kubectl get pods -n kube-system -w | grep coredns

# Restart both
kubectl -n ricplt rollout restart deployment/deployment-ricplt-a1mediator
kubectl -n ricplt rollout restart deployment/deployment-ricplt-e2mgr

# Wait for rollout to complete
kubectl -n ricplt rollout status deployment/deployment-ricplt-a1mediator --timeout=120s
kubectl -n ricplt rollout status deployment/deployment-ricplt-e2mgr --timeout=120s
