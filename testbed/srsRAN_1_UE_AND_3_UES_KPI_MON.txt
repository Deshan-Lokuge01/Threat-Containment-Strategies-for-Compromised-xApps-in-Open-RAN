04. Configure the route and test ping

sudo ip addr add 10.45.1.1/16 dev ogstun 2>/dev/null || true

UE1_IP=$(sudo ip netns exec ue1 \
ip -4 -o addr show tun_srsue |
awk '{print $4}' |
cut -d/ -f1)

sudo ip netns exec ue1 \
ip route replace 10.45.0.0/16 \
dev tun_srsue src "$UE1_IP"

sudo ip netns exec ue1 ping -c 5 -W 3 10.45.1.1

sudo ip netns exec ue1 ping -i 0.1 10.45.1.1

02. Fix 10.45.1.1 again after UE starts
Run this again after ue1 exists:

sudo ip addr add 10.45.1.1/16 dev ogstun 2>/dev/null || true

for ns in ue1 ue2 ue3; do
  ip=$(sudo ip netns exec "$ns" ip -4 -o addr show tun_srsue 2>/dev/null | awk '{print $4}' | cut -d/ -f1)
  if [ -n "$ip" ]; then
    sudo ip netns exec "$ns" ip route replace 10.45.0.0/16 dev tun_srsue src "$ip"
    echo "$ns route fixed with src $ip"
  fi
done
